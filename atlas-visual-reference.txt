function estimateTokens(value){
  if (Array.isArray(value)) return value.reduce((n,x)=>n+estimateTokens(x),0);
  if (value && typeof value === "object") {
    if (value.type === "image_url" || value.type === "input_image") return 2048;
    if (value.image_url || value.image) return 2048;
    return Object.entries(value).reduce((n,[k,v])=>n+estimateTokens(k)+estimateTokens(v),0);
  }
  const text=String(value ?? "");
  // Base64 image payload size is not proportional to model context tokens.
  if (/^data:image\//i.test(text)) return 2048;
  return Math.ceil(text.length / 3.2);
}
function trimToContext(messages, system, contextLimit, outputReserve){
  const budget = Math.max(1024, contextLimit - outputReserve - 1200);
  const kept = messages.slice();
  const systemTokens = estimateTokens(system) + 8;
  let count = systemTokens + kept.reduce((n,m)=>n+estimateTokens(m.content)+8,0);
  // Drop oldest complete history entries, but always preserve the newest user turn.
  while (count > budget && kept.length > 1) {
    const removed = kept.shift();
    count -= estimateTokens(removed.content) + 8;
  }
  return {messages:kept, estimated:count, budget, dropped:messages.length-kept.length};
}

/* ---------- markdown-lite ---------- */
function esc(s){ return String(s).replace(/[&<>"]/g, c => ({"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;"}[c])); }
function inline(s){
  return s.replace(/`([^`]+)`/g, "<code>$1</code>").replace(/\*\*([^*]+)\*\*/g, "<strong>$1</strong>")
          .replace(/\[([^\]]+)\]\((https?:[^)\s]+)\)/g, '<a href="$2" target="_blank" rel="noopener">$1</a>');
}
function md(src){
  let think = "";
  src = src.replace(/<think>([\s\S]*?)(<\/think>|$)/g, (_, t) => { think += t; return ""; });
  let html = "";
  esc(src).split(/```/).forEach((p, i) => {
    if (i % 2) { const nl = p.indexOf("\n"); html += `<pre><button class="btn copy" type="button">Copy</button><code>${nl >= 0 ? p.slice(nl + 1) : p}</code></pre>`; return; }
    let list = false;
    p.split("\n").forEach(line => {
      const h = line.match(/^(#{1,4})\s+(.*)/), li = line.match(/^\s*(?:[-*]|\d+\.)\s+(.*)/);
      if (li) { if (!list) { html += "<ul>"; list = true; } html += `<li>${inline(li[1])}</li>`; return; }
      if (list) { html += "</ul>"; list = false; }
      if (h) html += `<p><strong>${inline(h[2])}</strong></p>`; else if (line.trim()) html += `<p>${inline(line)}</p>`;
    });
    if (list) html += "</ul>";
  });
  if (think.trim()) html = `<details class="think"><summary>Thinking</summary>${esc(think).replace(/\n/g,"<br>")}</details>` + html;
  return html;
}

function atomicTopicTitle(text){

function atomicShiftAssessment(text,c){
  if(!c||c.messages.filter(m=>m.role==="user").length<2||c.atomicDisabled||ctrl)return {split:false,reason:"insufficient-context"};
  const fresh=atomicTerms(text), history=c.messages.slice(-10).map(m=>m.content||"").join(" "), old=atomicTerms([c.title,history].join(" "));
  const newFiles=atomicFileRefs(text), oldFiles=atomicFileRefs([c.title,history].join(" "));
  if(fresh.length<3&&!newFiles.length)return {split:false,reason:"short-or-vague"};
  const overlap=atomicSimilarity(fresh,old);
  const sharedFiles=newFiles.filter(f=>oldFiles.includes(f)).length;
  const novelFiles=newFiles.length>0&&sharedFiles===0;
  // File changes are a strong signal; vocabulary alone needs several distinct technical anchors.
  const technical=fresh.filter(w=>/[._/-]/.test(w)||/\d/.test(w)||/(api|mcp|css|html|python|rust|react|tauri|sqlite|postgres|docker|vulkan|llama|agent|server|client|frontend|backend)/.test(w));
  const lexicalShift=overlap<0.10&&technical.length>=2&&fresh.length>=5;
  const split=novelFiles||lexicalShift;
  return {split,reason:novelFiles?"different-file-scope":lexicalShift?"distinct-technical-topic":"related-topic",overlap,files:newFiles,terms:fresh};
}

  box.insertAdjacentHTML("beforeend", `<div class="stats">${esc(stats)}</div>`);
  c.messages.push({ role: "assistant", content: answer, reasoningSummary: thinking, model: b.model, stats }); save();
  $("#send").hidden = false; $("#stop").hidden = true; ctrl = null; poll(); renderQueue(); if(phase === "DONE") drainQueue(); return phase;
}

const splitThink = (_reasoning, raw) => {
  let a = raw, t = "";
  const marker = "[[ATLAS_REASONING_SUMMARY]]", start = a.indexOf(marker);
  if (start >= 0) {
    const tail = a.slice(start + marker.length), close = tail.indexOf("[[/ATLAS_REASONING_SUMMARY]]");
    t = (close >= 0 ? tail.slice(0, close) : tail).trim();
    a = a.slice(0, start).trimEnd();
  }
  a = a.replace(/<think>[\s\S]*?(?:<\/think>|$)/g, "").trimEnd();
  return [t, a.trim()];
};
async function stage(b, messages, title){  // one streamed model call shown as its own card
  const d = document.createElement("div"); d.className = "msg assistant stage";
  d.innerHTML = `<div class="who">${esc(title)} // ${esc(b.name.toUpperCase())} // ${esc(b.model.toUpperCase())}</div><div class="activity" role="status"></div><details class="think" open hidden><summary>WORKING // summary appears when ready</summary><div class="think-text"></div></details><div class="body"></div>`;
  const log = $("#log"), e = log.querySelector(".empty"); if (e) e.remove();
  log.appendChild(d); log.scrollTop = log.scrollHeight;
  const act = d.querySelector(".activity"), think = d.querySelector("details"), tt = d.querySelector(".think-text"), body = d.querySelector(".body");
  let reasoning = "", raw = "", stats = "", t0 = performance.now(), tFirst = 0, n = 0;
  let phase = status[b.model] === "ready" ? "READING" : "LOADING MODEL";
  const draw = cls => { act.className = "activity" + (cls ? " " + cls : ""); act.innerHTML = `<span class="pulse" aria-hidden="true"></span><b>${phase}</b><span>${((performance.now() - t0) / 1000).toFixed(1)} s${n ? ` // ${n} tokens` : ""}</span>`; };
  const timer = setInterval(() => { if (!tFirst) phase = status[b.model] === "ready" ? "READING" : "LOADING MODEL"; draw(); }, 250);
  const fp = setInterval(poll, 1000);
  const req = { model: b.model, stream: true, temperature: Number(b.temperature), top_p: Number(b.top_p), messages: [{ role: "system", content: b.system }, ...messages] };
  if (Number(b.max_tokens) > 0) req.max_tokens = Number(b.max_tokens);
  try {
    const r = await fetch("/v1/chat/completions", { method: "POST", headers: {"Content-Type":"application/json"}, body: JSON.stringify(req), signal: ctrl.signal });
    if (!r.ok) throw new Error((await r.text()).slice(0, 300));
    const rd = r.body.getReader(), dec = new TextDecoder(); let buf = "";
    for (;;) {
      const { value, done } = await rd.read(); if (done) break;
      buf += dec.decode(value, { stream: true }); const ls = buf.split("\n"); buf = ls.pop();
      for (const l of ls) {
        if (!l.startsWith("data:")) continue; const p = l.slice(5).trim(); if (p === "[DONE]") continue;
        try {
          const j = JSON.parse(p), dl = (j.choices && j.choices[0] && j.choices[0].delta) || {};
          if (dl.reasoning_content) { tFirst = tFirst || performance.now(); n++; phase = "THINKING"; }
          if (dl.content) { tFirst = tFirst || performance.now(); raw += dl.content; n++; }
          if (j.timings) stats = `${Math.round(j.timings.predicted_per_second * 10) / 10} tok/s // ${j.timings.predicted_n} tokens`;
        } catch {}
      }
      if (tFirst) clearInterval(fp);
      const [t, a] = splitThink(reasoning, raw);
      if (t) {
        think.hidden = false; tt.textContent = t; tt.scrollTop = tt.scrollHeight;
        think.querySelector("summary").textContent = "REASONING SUMMARY // click to expand";
      }
      if (a) { if (t) think.open = false; phase = "WRITING"; body.innerHTML = md(a); }
      log.scrollTop = log.scrollHeight;
    }
    clearInterval(timer); clearInterval(fp); phase = "DONE"; draw("done");
  } catch (err) {
    clearInterval(timer); clearInterval(fp);
    phase = err.name === "AbortError" ? "STOPPED" : "ERROR"; draw("err");
    if (err.name !== "AbortError") body.innerHTML = md("Error: " + err.message);
    throw err;
  }
  const [thinking, answer] = splitThink(reasoning, raw);
  if (!thinking) think.remove();
  if (stats) d.insertAdjacentHTML("beforeend", `<div class="stats">${esc(stats)}</div>`);
  return { answer, thinking, stats };
}
async function teamRun(goal, team, rounds){
  const planner = ensureBot("Agent planner", "tasks",
    "Break the goal into numbered steps. For each: the action, the tool or app, time in minutes, and a done-when check. Flag any step that sends, buys, deletes or publishes.",
    "Breaks goals into steps with time and checks.");
  const reviewer = ensureBot("Reviewer", "tasks",
    "You are a strict reviewer and tester. Check the work against its task: correctness, missing parts, bugs, edge cases, and whether tests cover it. First line must be PASS if it is ready, or FIX if not. After FIX, give a numbered list of exact changes. Be short.",
    "Checks each iteration. Replies PASS or a FIX list.");
  const c = { id: uid(), botId: S.bot, title: "TEAM: " + goal.slice(0, 34), messages: [] };
  S.chats.unshift(c); S.chat = c.id; save(); renderAll();
  const keep = (role, content, model, stats, reasoningSummary) => { c.messages.push({ role, content, model, stats, reasoningSummary }); save(); };
  addMsg("user", "TEAM GOAL: " + goal); keep("user", "TEAM GOAL: " + goal);
  ctrl = new AbortController(); $("#send").hidden = true; $("#stop").hidden = false; setStatus("TEAM RUN", "busy");
  try {
    const names = team.map(b => b.name);
    const p = await stage(planner, [{ role: "user", content:
`Goal: ${goal}

Team bots: ${names.join(", ")}.
Split the goal into 2 to 6 small steps in the order they must run. Give each step to the best bot from the team list.
Reply with ONLY a JSON array and no other text, for example:
[{"bot":"${names[0]}","task":"Write the function and unit tests for it"}]` }], "01 // PLAN");
    keep("assistant", p.answer, planner.model, p.stats, p.thinking);
    let steps = [];
    try { steps = JSON.parse((p.answer.match(/\[[\s\S]*\]/) || ["[]"])[0]); } catch {}
    steps = (Array.isArray(steps) ? steps : []).filter(s => s && s.task)
      .map(s => ({ bot: team.find(b => b.name.toLowerCase() === String(s.bot || "").toLowerCase()) || team[0], task: String(s.task) }));
    if (!steps.length) steps = team.map(b => ({ bot: b, task: goal }));  // plan unreadable: each bot takes a pass
    let carry = "";
    for (let i = 0; i < steps.length; i++) {
      const s = steps[i]; let work = "", fix = "";
      for (let it = 1; it <= rounds; it++) {
        const prompt = `Overall goal: ${goal}\n\nYour step (${i + 1} of ${steps.length}): ${s.task}\n\n`
          + (carry ? `Work handed to you from the previous step:\n${carry}\n\n` : "")
          + (work ? `Your previous attempt:\n${work}\n\nReviewer feedback to fix:\n${fix}\n\nReturn the full revised work, not only the changes.`
                  : "Do this step fully. If it is code, include tests and say how to run them.");
        const r = await stage(s.bot, [{ role: "user", content: prompt }], `${String(i + 2).padStart(2, "0")} // STEP ${i + 1}/${steps.length} ROUND ${it}`);
        work = r.answer; keep("assistant", work, s.bot.model, r.stats, r.thinking);
        const v = await stage(reviewer, [{ role: "user", content: `Task: ${s.task}\n\nWork to review:\n${work}` }], `REVIEW ${i + 1}.${it}`);
        keep("assistant", v.answer, reviewer.model, v.stats, v.thinking);
        if (/^\W*PASS/i.test(v.answer)) break;
        fix = v.answer;
      }
      carry = work;
    }
    setStatus("TEAM DONE", "ok");
  } catch (e) { setStatus(e.name === "AbortError" ? "STOPPED" : "ERROR", e.name === "AbortError" ? "ok" : "err"); }
  $("#send").hidden = false; $("#stop").hidden = true; ctrl = null; renderBots(); poll();
}
