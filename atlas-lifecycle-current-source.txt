/// Owns only the exact child processes spawned by this Atlas desktop instance.
///
/// The child registry uses a short-lived std mutex so shutdown never needs to
/// `try_lock()` a Tokio mutex that may be held across an HTTP readiness wait.
/// Startup itself is serialized separately with an async mutex.

    children: Mutex<Vec<Child>>,
    start_lock: AsyncMutex<()>,
    stop_requested: AtomicBool,
    started: AtomicBool,
}

fn port_ready(port: u16) -> bool {
    let address = format!("127.0.0.1:{port}");
    address
        .parse()
        .ok()
        .and_then(|addr| {
            std::net::TcpStream::connect_timeout(&addr, Duration::from_millis(250)).ok()
        })
        .is_some()
}

fn ensure_not_stopping(stop_requested: &AtomicBool) -> Result<(), String> {
    if stop_requested.load(Ordering::Acquire) {
        Err("Atlas startup was cancelled because shutdown was requested.".into())
    } else {
        Ok(())
    }
}

async fn wait_http(
    url: &str,
    seconds: u64,
    stop_requested: &AtomicBool,
) -> Result<(), String> {
    let client = reqwest::Client::builder()
        .timeout(Duration::from_secs(2))
        .build()
        .map_err(|e| e.to_string())?;
    let until = Instant::now() + Duration::from_secs(seconds);

    loop {
        ensure_not_stopping(stop_requested)?;
        if let Ok(response) = client.get(url).send().await {
            if response.status().is_success() {
                return ensure_not_stopping(stop_requested);
            }
        }
        if Instant::now() >= until {
            return Err(format!("Service did not become ready: {url}"));
        }
        sleep(Duration::from_millis(300)).await;
    }
}

fn hide_console_window(command: &mut Command) {
    #[cfg(windows)]
    {
        use std::os::windows::process::CommandExt;
        const CREATE_NO_WINDOW: u32 = 0x0800_0000;
        command.creation_flags(CREATE_NO_WINDOW);

fn spawn_python(root: &Path, script: &Path) -> Result<Child, String> {
    let stdout = log_file(root, "atlas-server.log")?;
    let stderr = stdout.try_clone().map_err(|e| e.to_string())?;
    let mut candidates: Vec<(&str, Vec<String>)> = vec![
        ("py", vec!["-3".into(), script.display().to_string()]),
        ("python", vec![script.display().to_string()]),
    ];
    #[cfg(not(windows))]
    {
        candidates = vec![
            ("python3", vec![script.display().to_string()]),
            ("python", vec![script.display().to_string()]),
        ];
    }

    let mut last = String::new();
    for (exe, args) in candidates {
        let mut command = Command::new(exe);
        command.args(args)
            .current_dir(root)
            .stdin(Stdio::null())
            .stdout(Stdio::from(stdout.try_clone().map_err(|e| e.to_string())?))
            .stderr(Stdio::from(stderr.try_clone().map_err(|e| e.to_string())?));
        hide_console_window(&mut command);
        match command.spawn() {
            Ok(child) => return Ok(child),
            Err(error) => last = format!("{exe}: {error}"),
        }
    }

    Err(format!(
        "Could not start Python server. Install Python 3 and ensure 'py' or 'python' is on PATH. {last}"
    ))
}

impl OwnedChildren {
    fn register(&self, child: Child) -> Result<(), String> {
        // Recover a poisoned mutex rather than dropping ownership of a live
        // child. The registry contains only Child handles created by Atlas.
        let mut children = self
            .children
            .lock()
            .unwrap_or_else(std::sync::PoisonError::into_inner);
        children.push(child);
        Ok(())
    }

    /// Serializes startup and rolls back every child started by this attempt if
    /// any later prerequisite, readiness check, or identity check fails.
    pub async fn start(&self) -> Result<String, String> {
        let _start_guard = self.start_lock.lock().await;

        // Startup can be requested by React lifecycle effects, the retry button,
        // or a future recovery path. Treat it as idempotent for this desktop
        // instance: never spawn another server after a successful start.
        if self.started.load(Ordering::Acquire) {
            return Ok("Atlas services are already running under this desktop instance.".into());
        }

        self.stop_requested.store(false, Ordering::Release);
        let result = self.start_inner().await;
        match result {
            Ok(message) if !self.stop_requested.load(Ordering::Acquire) => {
                self.started.store(true, Ordering::Release);
                Ok(message)
            }
            Ok(_) => {
                self.stop();
                Err("Atlas startup was cancelled because shutdown was requested.".into())
            }
            Err(error) => {
                // Startup failure must not leave llama-swap or the API orphaned.
                self.stop();
                Err(error)
            }
        }
    }

        {
            return Err(format!(
                "Port {api} is occupied by an Atlas API with unexpected configuration; expected server {api} and upstream {upstream}."
            ));
        }

        ensure_not_stopping(&self.stop_requested)?;
        Ok(format!(
            "Atlas services ready. API :{api}; llama-swap :{upstream}. Only child processes started by this desktop instance are owned by its supervisor."
        ))
    }

    /// Stops only processes retained in this instance's child registry.
    /// This is synchronous and does not silently skip shutdown because another
    /// async startup task happens to hold a Tokio mutex.
    pub fn stop(&self) {
        self.stop_requested.store(true, Ordering::Release);
        self.started.store(false, Ordering::Release);
        // Recover the registry even if a previous panic poisoned the mutex;
        // skipping cleanup would violate the owned-process lifecycle contract.
        let mut children = self
            .children
            .lock()
            .unwrap_or_else(std::sync::PoisonError::into_inner);

        for child in children.iter_mut().rev() {
            if child.try_wait().ok().flatten().is_none() {
                let _ = child.kill();
            }
            let _ = child.wait();
        }
        children.clear();
    }
}

                if self.accepted_task_types is not None:
                    claim_payload["task_types"]=self.accepted_task_types
                claimed=self._request(f"/api/ops/agents/{self.agent_id}/tasks/claim",claim_payload,True)
                task=claimed.get("task")
                if not task: self._stop.wait(self.poll_interval); continue
                tid=task["task_id"]
                trace_id=task.get("trace_id")
                try:
                    self.heartbeat("busy",tid,"executing")
                    result=self.execute(task)
                    body={"status":"completed","result":result if isinstance(result,dict) else {"value":result}}
                except Exception as exc:
                    body={"status":"failed","error":f"{type(exc).__name__}: {exc}","result":{}}
                body["trace_id"] = trace_id
                self._request(f"/api/ops/agents/{self.agent_id}/tasks/{tid}/result",body,True)
                self.heartbeat("online")
            except Exception as exc:
                try: self.event("agent.worker_error",str(exc),level="error")
                except Exception: pass
                self._stop.wait(min(10,self.poll_interval*2))
    def start(self):
        if self._thread and self._thread.is_alive(): return self
        if not self.agent_id: self.register()
        self._stop.clear(); self._thread=threading.Thread(target=self._run,name=f"atlas-agent-{self.name}",daemon=True); self._thread.start(); return self
    def stop(self,timeout=5):
        self._stop.set()
        if self._thread: self._thread.join(timeout)
        try: self.heartbeat("offline")
        except Exception: pass

#!/usr/bin/env python3
"""Atlas Swarm Programming Worker.

Consumes `programming` tasks from the Atlas Swarm API and writes generated
project files beneath a locally configured workspace root. It never executes
commands received from the server.
"""
from __future__ import annotations

import argparse
import hashlib
import json
import os
import re
import sys
import tempfile
import time
from pathlib import Path
from typing import Any

MAX_FILES = 250
MAX_FILE_BYTES = 2 * 1024 * 1024
MAX_TOTAL_BYTES = 20 * 1024 * 1024
MARKER = ".atlas-swarm-task.json"


def _safe_component(value: str) -> str:
    value = str(value or "").strip()
    value = re.sub(r"[^A-Za-z0-9._-]+", "-", value).strip(".-_")
    if not value or value in {".", ".."}:
        raise ValueError("project_name must contain letters or numbers")
    return value[:80]


def _safe_relative_path(value: str) -> Path:
    raw = str(value or "").replace("\\", "/").strip()
    if not raw or raw.startswith("/") or re.match(r"^[A-Za-z]:", raw):
        raise ValueError(f"File path must be relative: {value!r}")
    parts = raw.split("/")
    if any(part in {"", ".", ".."} for part in parts):
        raise ValueError(f"Unsafe relative file path: {value!r}")
    if any(":" in part for part in parts):
        raise ValueError(f"Invalid file path component: {value!r}")
    return Path(*parts)


def _normalise_files(value: Any) -> list[tuple[Path, bytes]]:
    if isinstance(value, dict):
        entries = [{"path": k, "content": v} for k, v in value.items()]
    elif isinstance(value, list):
        entries = value
    else:
        raise ValueError("payload.files must be an object or a list of {path, content}")

    if not entries or len(entries) > MAX_FILES:
        raise ValueError(f"Provide between 1 and {MAX_FILES} files")

    result: list[tuple[Path, bytes]] = []
    seen: set[str] = set()
    total = 0
    for item in entries:
        if not isinstance(item, dict) or not isinstance(item.get("content"), str):
            raise ValueError("Each file requires string fields: path and content")
        rel = _safe_relative_path(item.get("path"))
        key = rel.as_posix().casefold()
        if key == MARKER.casefold() or key in seen:
            raise ValueError(f"Duplicate or reserved file path: {rel.as_posix()}")
        seen.add(key)
        data = item["content"].encode("utf-8")
        if len(data) > MAX_FILE_BYTES:
            raise ValueError(f"{rel.as_posix()} exceeds the 2 MiB per-file limit")
        total += len(data)
        if total > MAX_TOTAL_BYTES:
            raise ValueError("Task exceeds the 20 MiB total output limit")
        result.append((rel, data))
    return result


def _atomic_write(destination: Path, data: bytes) -> None:
    destination.parent.mkdir(parents=True, exist_ok=True)
    # Refuse symlinks anywhere in the destination chain.
    current = destination.parent
    while current != current.parent:
        if current.is_symlink():
            raise ValueError(f"Refusing to write through symlink: {current}")
        current = current.parent
    if destination.exists() and destination.is_symlink():
        raise ValueError(f"Refusing to overwrite symlink: {destination}")
    fd, temp_name = tempfile.mkstemp(prefix=".atlas-write-", dir=str(destination.parent))
    try:
        with os.fdopen(fd, "wb") as stream:
            stream.write(data)
            stream.flush()
            os.fsync(stream.fileno())
        os.replace(temp_name, destination)
    finally:
        try:
            os.unlink(temp_name)
        except FileNotFoundError:
            pass


def execute_programming_task(task: dict[str, Any], workspace_root: Path) -> dict[str, Any]:
    if not isinstance(task, dict):
        raise ValueError("Swarm task must be an object")
    task_type = str(task.get("task_type") or task.get("type") or "")
    if task_type != "programming":
        raise ValueError(f"Unsupported task type: {task_type!r}; this worker accepts 'programming'")

    payload = task.get("payload")
    if not isinstance(payload, dict):
        raise ValueError("Programming task payload must be an object")

    task_id = str(task.get("task_id") or "").strip()
    if not re.fullmatch(r"[A-Za-z0-9_-]{1,128}", task_id):
        raise ValueError("Task is missing a valid task_id")

    project_name = _safe_component(payload.get("project_name") or "atlas-programming-task")
    task_folder = f"{project_name}-{task_id[-12:]}"
    root = workspace_root.expanduser().resolve()
    root.mkdir(parents=True, exist_ok=True)
    if root.is_symlink():
        raise ValueError("Workspace root cannot be a symlink")

    target = (root / task_folder).resolve()
    if target.parent != root:
        raise ValueError("Resolved task folder escaped the configured workspace root")

    marker = target / MARKER
    if target.exists():
        if not marker.is_file():
            raise FileExistsError(f"Refusing to reuse existing folder without Atlas task marker: {target}")
        existing = json.loads(marker.read_text(encoding="utf-8"))
        if existing.get("task_id") != task_id:
            raise FileExistsError(f"Task folder belongs to a different task: {target}")
    else:
        target.mkdir(parents=False, exist_ok=False)

    files = _normalise_files(payload.get("files"))
    written = []
    digest = hashlib.sha256()
    for relative, data in files:
        destination = target / relative
        resolved_parent = destination.parent.resolve()
        if resolved_parent != target and target not in resolved_parent.parents:
            raise ValueError(f"File path escaped task folder: {relative}")
        _atomic_write(destination, data)
        digest.update(relative.as_posix().encode("utf-8"))
        digest.update(b"\0")
        digest.update(data)
        written.append({"path": relative.as_posix(), "bytes": len(data)})

    manifest = {
        "task_id": task_id,
        "project_name": project_name,
        "created_by": "atlas-swarm-programming-worker",
        "files": written,
        "sha256": digest.hexdigest(),
        "updated_at": time.time(),
    }
    _atomic_write(marker, json.dumps(manifest, ensure_ascii=False, indent=2).encode("utf-8"))
    return {
        "ok": True,
        "task_id": task_id,
        "workspace": str(target),
        "files_written": len(written),
        "total_bytes": sum(item["bytes"] for item in written),
        "sha256": manifest["sha256"],
        "executed_commands": 0,
    }


def self_test() -> int:
    import shutil

    root = Path(tempfile.gettempdir()) / f"atlas-swarm-worker-test-{uuid.uuid4().hex[:8]}"

def main() -> int:
    parser = argparse.ArgumentParser(description="Atlas Swarm programming worker")
